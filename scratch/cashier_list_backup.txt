const MUTED_SCHEMES = [
    { dot: '#818cf8', ring: '#e0e7ff', text: '#4338ca', bg: '#eef2ff' }, // indigo
    { dot: '#60a5fa', ring: '#dbeafe', text: '#1d4ed8', bg: '#eff6ff' }, // blue
    { dot: '#34d399', ring: '#d1fae5', text: '#065f46', bg: '#ecfdf5' }, // emerald
    { dot: '#a78bfa', ring: '#ede9fe', text: '#5b21b6', bg: '#f5f3ff' }, // violet
    { dot: '#f9a8d4', ring: '#fce7f3', text: '#9d174d', bg: '#fdf2f8' }, // pink
    { dot: '#67e8f9', ring: '#cffafe', text: '#0e7490', bg: '#ecfeff' }, // cyan
];
const scheme = (name = '') => MUTED_SCHEMES[name.charCodeAt(0) % MUTED_SCHEMES.length];
const initials = (name = '') => name.slice(0, 2).toUpperCase() || '??';
const mockStat = (name = '') => {
    const s = name.charCodeAt(0) || 65;
    const r = 70 + (s % 28);
    const label = r >= 90 ? 'Excelente' : r >= 80 ? 'Bueno' : r >= 70 ? 'Regular' : 'Bajo';
    const dotColor = r >= 90 ? '#10b981' : r >= 80 ? '#60a5fa' : r >= 70 ? '#fbbf24' : '#f87171';
    return { att: 38 + (s % 55), rend: r, sat: (6.8 + ((s % 32) / 10)).toFixed(1), label, dotColor };
};
